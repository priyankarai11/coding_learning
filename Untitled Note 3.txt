    if (str.length <= 1) return [str];
return str.split('').flatMap((n,i)=>{
    const remaining=str.slice(0,i)+str.slice(i+1)
    return permute(remaining).map(ele=>n+ele)
})
}

function permute(str) {
    if (str.length <= 1) return [str];
const res=[]
    for(let i=0;i<str.length;i++){
        const char=str[i]
        const rem=str.slice(0,i)+str.slice(i+1)
        for(let perm of permute(rem)){
            res.push(char+perm)
        }
    }
    return res
}

function matchesABC(str) {
    let i = 0;

    while (i < str.length && str[i] === 'a') i++;
    while (i < str.length && str[i] === 'b') i++;
    while (i < str.length && str[i] === 'c') i++;

    return i === str.length;
}

function countSubstringsWithAllVowels(str) {
    const vowels = new Set(['a', 'e', 'i', 'o', 'u']);
    let count = 0;

    for (let i = 0; i < str.length; i++) {
        let found = new Set();
        for (let j = i; j < str.length; j++) {
            if (vowels.has(str[j])) {
                found.add(str[j]);
            }
            if (found.size === 5) { // all vowels found
                count++;
            }
        }
    }

    return count;
}

console.log(countSubstringsWithAllVowels("aeiou"));         // 1
console.log(countSubstringsWithAllVowels("aeiouxyzaeiou")); // 3+

Find the longest substring with even counts of all characters
const str = "abba";
let longest = "";

for (let i = 0; i < str.length; i++) {
    let freq = {};
    for (let j = i; j < str.length; j++) {
        let char = str[j];
        freq[char] = (freq[char] || 0) + 1;

        // Check if all counts are even
        let allEven = true;
        for (let key in freq) {
            if (freq[key] % 2 !== 0) {
                allEven = false;
                break;
            }
        }

        // Update longest if valid
        if (allEven && (j - i + 1) > longest.length) {
            longest = str.slice(i, j + 1);
        }
    }
}

console.log(longest); // "abba"
