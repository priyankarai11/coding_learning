function isoTrophic(s,t){
    for(let i=0;i<s.length;i++){
        if(s.indexOf(s[i])!==t.indexOf(t[i])){
            return false
        }
    }
    return true
}

3. Longest Palindromic Substring

let res=''
function extend(left,right){
    while(left>=0 && right<=str.length-1 && str[left]===str[right]){
        left--
        right++
    }
    return str.slice(left+1,right)
}


for(let i=0;i<str.length;i++){
    let odd=extend(i,i)
    let even=extend(i,i+1)
    if(odd.length>res.length) res=odd
    if(even.length>res.length) res=even
}

6. Check Balanced Parentheses

function check(str){
    let s=[], map={')':'(','}':'{',']':'['}
    for(let i of str){
        if(i==='(' || i==='[' || i==='{'){
            s.push(i)
        }else{
            if(s.pop()!==map[i]){
                return false;
            }
        }
    }
    return s.length===0
}
console.log(check('{[()]}}'))

9. Find All Permutations of String

function perm(str){
    if(str.length===0) return [str]
    
    return str.split('').flatMap((char,i)=>{
        const replaced=str.slice(0,i)+str.slice(i+1)
        return perm(replaced).map(ele=>char+ele)
    })
}

console.log(perm("abc"))


function perm(str){
    if(str.length===0) return [str]
    let res=[]
    for(let i=0;i<str.length;i++){

        const replaced=str.slice(0,i)+str.slice(i+1)
        
        for(let ele of perm(replaced)){
            res.push(char+ele)
        }
    }
        
     return res  
}

11. Count Substrings with Exactly K Distinct Characters

function countAtMostK(s,k){
    let count={}
    let left=0;res=0
    
    for(let right=0;right<s.length;right++){
        count[s[right]]=(count[s[right]]||0)+1
        
        while(Object.keys(count).length>k){
            count[s[left]]--;
            if(count[s[left]]===0) delete count[s[left]];
            left++
        }
        
        res+=right-left+1
    }
    return res
}
let str = "pqpqs", k = 2;
console.log(countAtMostK(str, k) - countAtMostK(str, k - 1))

let str = "cbbcc";

    const copy=[...str]
    copy.splice(i,1)
    const c=copy.join('')
    const rev=[...copy].reverse().join('')

        found=true
        break
    }
}

15. Decode Encoded String (LeetCode 394 style)

Input: "3[a]2[bc]" → Output: "aaabcbc"

const str = "3[a]2[bc]3[priya]";
let res = '';
const pattern = /(\d+)\[([a-z]+)\]/g; // matches number[someletters]
let match;

// loop through all matches
while ((match = pattern.exec(str)) !== null) {
    const num = Number(match[1]);  // the number before brackets
    const letters = match[2];      // letters inside brackets
    res += letters.repeat(num);    // repeat letters and append to result
}

console.log(res); // "aaabcbc"

const res = str.replace(/(\d+)\[([a-z]+)\]/g, (_, num, letters) => letters.repeat(num));

console.log(res);

17. remove string with proper index

const s1 = "I choose a Lazy person to do a hard job"; 
const s2 = "Choose a person a  hard";

let arr1= s1.split(" ").map(each=>each.toLowerCase())
let arr2= s2.split(" ").map(each=>each.toLowerCase())
// console.log(arr1,arr2)
const indexToBeDeleted=[]
let i=0;
let j=0;

while(j!=arr2.length)
{
        if(i==arr1.length){
            i=0
        }
        if(arr1[i]==arr2[j])
        {
            indexToBeDeleted.push(i)
            i++
            j++
        }
        else{
            i++
        }
}

let res=s1.split(' ').filter((_,index)=>!indexToBeDeleted.includes(index)).join(' ')
