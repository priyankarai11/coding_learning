Array.prototype.mapPolyfill = function(callback) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    result.push(callback(this[i], i, this));
  }
  return result;
};

const doubled = arr.mapPolyfill(x => x * 2);

function flatten(input) {
  let result = [];
  for (let i of input) {
    if (Array.isArray(i)) {
      result = result.concat(flatten(i)); // recursive flatten
    } else {
      result.push(i);
    }
  }
  return result;
}

const res=[]
for(let i=0;i<arr.length-2;i++){
    if(i>0 && arr[i]===arr[i-1])  continue;
    let first=i+1,second=arr.length-1;
    while(first<second){
    let sum=arr[i]+arr[first]+arr[second];
    if(sum===0){
        res.push([arr[i],arr[first],arr[second]])
        while(arr[first]===arr[first+1]) first++;
        while(arr[second]===arr[second-1]) second--;
        first++; second--;
    }else{
        sum<0? first++:second--;
    }
  }
}
